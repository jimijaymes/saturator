cmake_minimum_required(VERSION 3.22)
project(Saturator VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(Saturator COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.saturator.v02" PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE St02 FORMATS VST3 PRODUCT_NAME "Saturator 0.2")
juce_generate_juce_header(Saturator)
target_sources(Saturator PRIVATE Source/PluginProcessor.h Source/PluginProcessor.cpp Source/PluginEditor.h Source/PluginEditor.cpp)
target_compile_definitions(Saturator PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Saturator PRIVATE juce::juce_audio_utils juce::juce_dsp)
